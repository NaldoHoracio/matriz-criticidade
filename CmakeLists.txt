cmake_minimum_required(VERSION 3.16)

project(MatrizCriticidade VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)

find_package(Qt6 REQUIRED COMPONENTS Quick Test)

qt_standard_project_setup()

include_directories(header)

qt_add_executable(appMatriz
    src/main.cpp
    src/AppController.cpp
)

qt_add_qml_module(appMatriz
    URI MatrizCriticidade
    VERSION 1.0
    QML_FILES
        qml/Main.qml
)

qt_add_resources(appMatriz "resources"
    PREFIX "/"
    FILES resources.qrc
)

target_link_libraries(appMatriz
    PRIVATE Qt6::Quick
)

enable_testing()
add_subdirectory(tests)
